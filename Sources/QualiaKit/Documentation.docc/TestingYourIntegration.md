# Testing your integration

Add QualiaTesting only to test/demo targets. FixtureAnalyzer indexes immutable
observations by input ID and fails when an entry is absent; it never interprets
text. QualiaSessionHarness combines the real session with TestQualiaClock and
RecordingHapticRenderer. The renderer records ordered descriptors and ownership
without importing Core Haptics in the consumer or claiming physical sensation.

## Scripted LivingStory

<!-- snippet: Examples/DeveloperExperience/Sources/QualiaExamples/LivingStory.swift#living-story -->
```swift
import QualiaKit
import QualiaTesting

/// Scripted inputs demonstrate orchestration; no text meaning is inferred.
public enum LivingStory {
    public static func analyzer() throws -> FixtureAnalyzer {
        let english = try QualiaLanguage(rawValue: "en")
        let identity = try QualiaAnalyzerIdentity(identifier: "example.scripted-story", version: "1")
        return try FixtureAnalyzer(
            capabilities: .init(languages: [english], dimensions: [],
                signals: [.suspense, .threat, .impact], acceptsContext: true, execution: .onDevice),
            observations: [
                .init(inputID: .init(rawValue: "build"),
                      signals: [.suspense: .init(value: 0.5, confidence: 1)], language: english, analyzer: identity),
                .init(inputID: .init(rawValue: "impact"),
                      signals: [.suspense: .init(value: 1, confidence: 1), .impact: .init(value: 1, confidence: 1)],
                      language: english, analyzer: identity),
            ]
        )
    }

    public static func configuration() throws -> QualiaSessionConfiguration {
        try .init(language: .init(policy: .requireExplicit, minimumConfidence: 0.75),
            context: .init(maximumFragments: 4, maximumCharacters: 1_000), sceneState: .narrative,
            reaction: .horrorNarrative(.init(heartbeat: .init(
                startThreshold: 0.5, stopThreshold: 0.25, minimumBPM: 60, maximumBPM: 120))))
    }

    public static func input(_ id: String) throws -> QualiaInput {
        try .init(id: .init(rawValue: id), text: "Repository-authored scripted scene.", language: .init(rawValue: "en"))
    }

    @MainActor
    public static func makeSession(renderer: any HapticRendering,
                                   clock: any QualiaClock = QualiaContinuousClock()) async throws -> QualiaSession {
        try await QualiaSession.configured(analyzer: analyzer(), renderer: renderer,
                                         configuration: configuration(), clock: clock)
    }
}
```

The following complete test compiles in an external package using public imports.
The descriptor expectations are literal hand-calculated values, not generated by
the policy under test. Only the newly allocated session owner is substituted.

<!-- snippet: Examples/DeveloperExperience/Tests/ConsumerTests/DeveloperExperienceTests.swift#integration-test -->
```swift
import XCTest
import QualiaKit
import QualiaTesting
import QualiaExamples

@MainActor
final class LivingStoryIntegrationTests: XCTestCase {
    func testLivingStoryOrderedCommandsAndReset() async throws {
        let harness = try await QualiaSessionHarness(
            analyzer: LivingStory.analyzer(), configuration: LivingStory.configuration()
        )
        _ = try await harness.process(LivingStory.input("build"))
        _ = try await harness.process(LivingStory.input("impact"), advancingBy: .seconds(1))
        try await harness.reset()

        // Independent literal descriptors: no policy or pattern factory is the oracle.
        let id = try HapticEffectID(rawValue: "qualia.horror-narrative.heartbeat", scope: .owned(harness.session.owner))
        let initial = try HapticPattern(duration: .seconds(2.0 / 3), events: [
            .transient(at: .zero, intensity: HapticValue(0.4), sharpness: HapticValue(0.2)),
            .transient(at: .seconds(1.0 / 6), intensity: HapticValue(0.2800000011920929), sharpness: HapticValue(0.2)),
        ], looping: .loop(period: .seconds(2.0 / 3)), playbackDuration: .seconds(12))
        let updated = try HapticPattern(duration: .seconds(0.5423728813559322), events: [
            .transient(at: .zero, intensity: HapticValue(0.5718749761581421), sharpness: HapticValue(0.2)),
            .transient(at: .seconds(0.13559322033898305), intensity: HapticValue(0.40031248331069946), sharpness: HapticValue(0.2)),
        ], looping: .loop(period: .seconds(0.5423728813559322)), playbackDuration: .seconds(11))
        let accent = try HapticPattern(duration: .milliseconds(120), events: [
            .transient(at: .zero, intensity: HapticValue(1), sharpness: HapticValue(0.85))
        ])
        XCTAssertEqual(harness.renderer.commands, [
            .start(id: id, pattern: initial, channel: .ambient),
            .replace(id: id, pattern: updated, channel: .ambient),
            .play(pattern: accent, channel: .accent), .stop(id: id),
        ])
        XCTAssertTrue(harness.renderer.activeEffects.isEmpty)
    }
}
```

Run `swift test --package-path Examples/DeveloperExperience`. The other consumer
tests pair a custom analyzer/policy and exercise the redacted diagnostic report.
For concurrency tests, ControlledAnalyzer exposes request numbers and completion
barriers; explicitly choose whether it respects cancellation. Advance the test
clock after waiters are registered. Do not replace barriers with arbitrary sleeps.
Call `cancelAll()` in teardown to release controlled requests/clock sleepers.

This verifies runtime behavior. It does not measure model accuracy, comfort or
perceived narrative alignment. Use separately annotated data and physical-device
observations for those claims. <doc:Performance> describes measurement
limits and <doc:PrivacyAndDataFlow> explains synthetic fixture requirements.
